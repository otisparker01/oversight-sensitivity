```python
from typing import List


def paint_walls(cost: List[int], time: List[int]) -> int:
    """ You have two painters to paint n walls: a paid painter and a free painter.
    The paid painter paints wall i in time[i] units and costs cost[i] money.
    The free painter paints any wall in 1 unit of time for free, but can only work
    while the paid painter is busy.
    
    Return the minimum cost to paint all n walls.
    
    >>> paint_walls([1, 2, 3, 2], [1, 2, 3, 2])
    3
    >>> paint_walls([2, 3, 4, 2], [1, 1, 1, 1])
    4
    """
    n = len(cost)
    # DP: dp[w] = minimum cost to achieve total weight >= w
    # weight of wall i = 1 + time[i]
    # We need weight >= n
    INF = 10**15
    
    # We only need to track up to n (cap at n)
    dp = [INF] * (n + 1)
    dp[0] = 0
    
    for i in range(n):
        w = 1 + time[i]
        c = cost[i]
        for j in range(n, -1, -1):
            if dp[j] != INF:
                new_weight = min(n, j + w)
                if dp[j] + c < dp[new_weight]:
                    dp[new_weight] = dp[j] + c
    
    return dp[n]
```

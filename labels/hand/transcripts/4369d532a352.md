```python
from typing import List


def _longest_distinct_pairs(pair_vals):
    """Find longest subarray of non-None values with all distinct values.
    Returns the maximum number of consecutive valid pairs."""
    n = len(pair_vals)
    left = 0
    max_pairs = 0
    # last_seen[value] = most recent index where value was seen
    last_seen = {}
    
    for right in range(n):
        val = pair_vals[right]
        if val is None:
            # Invalid pair acts as barrier: reset window after this position
            left = right + 1
            last_seen.clear()
            continue
        
        if val in last_seen and last_seen[val] >= left:
            # Duplicate value: shrink window from left
            left = last_seen[val] + 1
        
        last_seen[val] = right
        cur_pairs = right - left + 1
        if cur_pairs > max_pairs:
            max_pairs = cur_pairs
    
    return max_pairs


def max_1122_subarray_length(n: int, a: List[int]) -> int:
    """ Given a list of positive integers, find the maximum length of a contiguous subarray 
    that forms a 1122 sequence.
    
    A 1122 sequence is defined as a sequence that:
    1. Has even length
    2. For each i from 1 to len/2, elements at positions 2i-1 and 2i are equal (1-indexed)
    3. Each positive integer appears either 0 times or exactly 2 times in the sequence
    
    Args:
        n: The length of the array (1 <= n <= 2*10^5)
        a: The array of positive integers (1 <= a[i] <= n)
    
    Returns:
        The maximum length of a contiguous subarray that is a 1122 sequence
    
    >>> max_1122_subarray_length(8, [2, 3, 1, 1, 2, 2, 1, 1])
    4
    >>> max_1122_subarray_length(3, [1, 2, 2])
    2
    >>> max_1122_subarray_length(1, [1])
    0
    """
    best = 0
    
    # Try both parities: subarrays starting at even indices (0, 2, 4, ...)
    # and subarrays starting at odd indices (1, 3, 5, ...)
    for parity in (0, 1):
        pair_vals = []
        k = 0
        while 2 * k + parity + 1 < n:
            left_idx = 2 * k + parity
            right_idx = left_idx + 1
            if a[left_idx] == a[right_idx]:
                pair_vals.append(a[left_idx])
            else:
                pair_vals.append(None)
            k += 1
        
        max_pairs = _longest_distinct_pairs(pair_vals)
        cur_len = max_pairs * 2
        if cur_len > best:
            best = cur_len
    
    return best
```

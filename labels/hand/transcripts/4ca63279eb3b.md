def find_cube_difference_pair(n: int) -> Optional[Tuple[int, int]]:
    """ Find a pair of positive integers (x, y) such that x^3 - y^3 = n.
    
    Args:
        n: A positive integer (1 <= n <= 10^18)
    
    Returns:
        A tuple (x, y) of positive integers such that x^3 - y^3 = n if such a pair exists,
        None otherwise.
    
    >>> find_cube_difference_pair(397)
    (12, 11)
    >>> find_cube_difference_pair(1)
    None
    >>> find_cube_difference_pair(7)
    (2, 1)
    """
    if n <= 0:
        return None

    # x^3 - y^3 = (x - y)(x^2 + xy + y^2) = d * (x^2 + xy + y^2) = n
    # Let d = x - y > 0, then d divides n and d^3 < n.
    # x = y + d, substitute:
    # 3d*y^2 + 3d^2*y + (d^3 - n) = 0
    # Solve for y: y = (-3d^2 + sqrt(12dn - 3d^4)) / (6d)

    # Integer cube root to bound d
    max_d = 1
    while max_d ** 3 <= n:
        max_d += 1

    for d in range(1, max_d + 1):
        if n % d != 0:
            continue

        # Discriminant: Δ = 12*d*n - 3*d^4
        delta = 12 * d * n - 3 * d * d * d * d
        if delta < 0:
            continue

        s = math.isqrt(delta)
        if s * s != delta:
            continue

        # y = (s - 3*d^2) / (6*d)
        numerator = s - 3 * d * d
        if numerator <= 0:
            continue
        if numerator % (6 * d) != 0:
            continue

        y = numerator // (6 * d)
        x = y + d

        if x > 0 and y > 0 and x ** 3 - y ** 3 == n:
            return (x, y)

    return None

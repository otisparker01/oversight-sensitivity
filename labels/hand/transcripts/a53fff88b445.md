from collections import defaultdict, deque
from typing import List


def min_ng_list_size(n: int, product_names: List[str]) -> int:
    """ Given a list of product names (each consisting of exactly 2 uppercase letters),
    find the minimum number of strings needed in an NG list such that:
    - Each product name appears as a substring in at least one string in the list
    - No string in the list contains any 2-letter substring that is not a product name
    
    Strings can be formed by chaining product names that share a letter 
    (e.g., "AB" and "BC" can form "ABC").
    
    Args:
        n: Number of product names (1 <= n <= 26^2)
        product_names: List of n distinct strings, each of length 2 with uppercase letters
    
    Returns:
        The minimum number of strings needed in the NG list
    
    >>> min_ng_list_size(7, ['AB', 'BC', 'CA', 'CD', 'DE', 'DF', 'XX'])
    3
    >>> min_ng_list_size(5, ['AC', 'BC', 'CD', 'DE', 'DF'])
    2
    >>> min_ng_list_size(6, ['AB', 'AC', 'CB', 'AD', 'DB', 'BA'])
    1
    """
    # Build graph: vertices are uppercase letters, edges are product names
    outdeg = defaultdict(int)
    indeg = defaultdict(int)
    adj = defaultdict(set)          # directed adjacency
    undirected_adj = defaultdict(set)
    vertices: set[str] = set()
    
    for name in product_names:
        u, v = name[0], name[1]
        outdeg[u] += 1
        indeg[v] += 1
        adj[u].add(v)
        undirected_adj[u].add(v)
        undirected_adj[v].add(u)
        vertices.add(u)
        vertices.add(v)
    
    # Net imbalance at each vertex
    d = {v: outdeg[v] - indeg[v] for v in vertices}
    
    # Find weakly connected components via undirected adjacency
    visited: set[str] = set()
    components: list[set[str]] = []
    for v in vertices:
        if v not in visited:
            stack = [v]
            visited.add(v)
            comp: set[str] = set()
            while stack:
                node = stack.pop()
                comp.add(node)
                for nb in undirected_adj[node]:
                    if nb not in visited:
                        visited.add(nb)
                        stack.append(nb)
            components.append(comp)
    
    total_walks = 0
    
    for comp in components:
        # Separate surplus (d>0) and deficit (d<0) vertices
        surplus_vertices = [(v, d[v]) for v in comp if d[v] > 0]
        deficit_vertices = [(v, -d[v]) for v in comp if d[v] < 0]
        
        total_surplus = sum(amt for _, amt in surplus_vertices)
        
        if total_surplus == 0:
            # Balanced component: one walk suffices
            total_walks += 1
            continue
        
        # Compute reachability from each deficit vertex to each surplus vertex
        # (flow goes deficit -> surplus along directed edges)
        reachable: dict[tuple[str, str], bool] = {}
        for t, _ in deficit_vertices:
            seen: set[str] = set()
            q = deque([t])
            seen.add(t)
            while q:
                node = q.popleft()
                for nb in adj[node]:
                    if nb not in seen:
                        seen.add(nb)
                        q.append(nb)
            for s, _ in surplus_vertices:
                if s in seen:
                    reachable[(t, s)] = True
        
        # Build flow network and compute max flow
        S = len(surplus_vertices)
        T = len(deficit_vertices)
        N = 2 + S + T
        source = 0
        sink = N - 1
        
        cap = [[0] * N for _ in range(N)]
        
        for i, (_, amt) in enumerate(deficit_vertices):
            cap[source][1 + i] = amt
        
        for j, (_, amt) in enumerate(surplus_vertices):
            cap[1 + T + j][sink] = amt
        
        for i, (t, _) in enumerate(deficit_vertices):
            for j, (s, _) in enumerate(surplus_vertices):
                if (t, s) in reachable:
                    cap[1 + i][1 + T + j] = 10 ** 9
        
        # Ford-Fulkerson max flow
        def bfs_flow() -> list[int] | None:
            parent = [-1] * N
            parent[source] = source
            q = deque([source])
            while q:
                u = q.popleft()
                for v in range(N):
                    if parent[v] == -1 and cap[u][v] > 0:
                        parent[v] = u
                        if v == sink:
                            return parent
                        q.append(v)
            return None
        
        max_flow = 0
        while True:
            parent = bfs_flow()
            if parent is None:
                break
            # Find bottleneck
            v = sink
            flow = float('inf')
            while v != source:
                u = parent[v]
                flow = min(flow, cap[u][v])
                v = u
            # Augment
            v = sink
            while v != source:
                u = parent[v]
                cap[u][v] -= flow
                cap[v][u] += flow
                v = u
            max_flow += flow
        
        # Minimum walks for this component
        comp_walks = max(1, total_surplus - max_flow)
        total_walks += comp_walks
    
    return total_walks

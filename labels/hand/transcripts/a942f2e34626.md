```python
from typing import List, Tuple


def sum_absolute_differences(n: int, a: List[int], b: List[int], queries: List[Tuple[int, int]]) -> List[int]:
    """ Given two arrays A and B of length n, and a list of queries where each query is a tuple (x, y),
    compute for each query the sum of |A[i] - B[j]| for all i from 0 to x-1 and j from 0 to y-1.
    
    Args:
        n: The length of arrays A and B (1 <= n <= 10^5)
        a: Array A with elements (0 <= A[i] <= 2*10^8)
        b: Array B with elements (0 <= B[j] <= 2*10^8)
        queries: List of tuples (x, y) where 1 <= x, y <= n
    
    Returns:
        A list of integers where the k-th element is the sum for the k-th query
    
    >>> sum_absolute_differences(2, [2, 4], [3, 5], [(1, 1), (1, 2), (2, 1), (2, 2)])
    [1, 4, 2, 6]
    >>> sum_absolute_differences(5, [1163686, 28892, 1263085, 2347878, 520306], [1332157, 1202905, 2437161, 1291976, 563395], [(5, 3), (1, 5), (2, 3), (1, 2), (5, 5)])
    [13331322, 2209746, 6366712, 207690, 20241215]
    """
    results = []
    
    for x, y in queries:
        # Extract prefixes and sort them
        a_prefix = sorted(a[:x])
        b_prefix = sorted(b[:y])
        
        # Compute sum of absolute differences using prefix sums on sorted B
        # For each element in a_prefix, find split point in b_prefix
        pref_b_sorted = [0] * (y + 1)
        for j in range(y):
            pref_b_sorted[j + 1] = pref_b_sorted[j] + b_prefix[j]
        
        total = 0
        # Two-pointer approach: iterate through a_prefix, maintain split in b_prefix
        b_idx = 0  # first index where b_prefix[b_idx] > a_prefix[i]
        for a_val in a_prefix:
            # Advance b_idx while b_prefix[b_idx] <= a_val
            while b_idx < y and b_prefix[b_idx] <= a_val:
                b_idx += 1
            # b_idx is the count of B elements <= a_val
            # For B elements <= a_val: contribution = a_val - B[j]
            # Sum over these: b_idx * a_val - pref_b_sorted[b_idx]
            left_sum = b_idx * a_val - pref_b_sorted[b_idx]
            # For B elements > a_val: contribution = B[j] - a_val
            # Sum over these: (pref_b_sorted[y] - pref_b_sorted[b_idx]) - (y - b_idx) * a_val
            right_sum = (pref_b_sorted[y] - pref_b_sorted[b_idx]) - (y - b_idx) * a_val
            total += left_sum + right_sum
        
        results.append(total)
    
    return results
```

def sum_good_sequence_scores(n: int, a: list[int]) -> int:
    """ Given a positive integer n and a list a of length n-1 containing positive integers,
    find the sum of scores of all good sequences modulo 998244353.
    
    A sequence S of length n is called "good" if:
    - For every i from 0 to n-2, f(S[i]/S[i+1]) = a[i], where f(p/q) = p*q when gcd(p,q) = 1
    - gcd of all elements in S equals 1
    
    The score of a sequence is the product of all its elements.
    
    Args:
        n: The length of the sequences (2 <= n <= 1000)
        a: List of n-1 positive integers (1 <= a[i] <= 1000)
    
    Returns:
        The sum of scores of all good sequences modulo 998244353
    
    >>> sum_good_sequence_scores(6, [1, 9, 2, 2, 9])
    939634344
    >>> sum_good_sequence_scores(2, [9])
    18
    >>> sum_good_sequence_scores(3, [1, 1])
    1
    """
    max_val = max(a) if a else 0
    
    # Sieve primes up to max_val
    is_prime = [True] * (max_val + 1)
    if max_val >= 0:
        is_prime[0] = False
    if max_val >= 1:
        is_prime[1] = False
    primes = []
    for i in range(2, max_val + 1):
        if is_prime[i]:
            primes.append(i)
            step = i
            start = i * i
            for j in range(start, max_val + 1, step):
                is_prime[j] = False
    
    ans = 1
    for p in primes:
        # Compute alpha: exponents of p in each a[i]
        alpha = [0] * (n - 1)
        present = False
        for idx, val in enumerate(a):
            e = 0
            while val % p == 0:
                val //= p
                e += 1
            alpha[idx] = e
            if e > 0:
                present = True
        
        if not present:
            continue
        
        S = sum(alpha)
        if S == 0:
            # Shouldn't happen since present=True means some alpha_i > 0
            continue
        
        # Precompute powers of p mod MOD
        pow_p = [1] * (S + 1)
        for i in range(1, S + 1):
            pow_p[i] = (pow_p[i - 1] * p) % MOD
        
        # dp0[x] = sum of scores for paths ending at current pos with value x, min>0 (no zero yet)
        # dp1[x] = same but min==0 (have seen zero)
        dp0 = [0] * (S + 1)
        dp1 = [0] * (S + 1)
        
        # Initialize for position 0: x_0 can be any value in [0, S]
        for x in range(S + 1):
            val = pow_p[x]  # p^{x_0}
            if x == 0:
                dp1[x] = val
            else:
                dp0[x] = val
        
        # Process transitions for positions i = 0..n-2
        for ai in alpha:
            new0 = [0] * (S + 1)
            new1 = [0] * (S + 1)
            
            if ai == 0:
                for x in range(S + 1):
                    w0 = dp0[x]
                    w1 = dp1[x]
                    if w0 == 0 and w1 == 0:
                        continue
                    mul = pow_p[x]  # p^{x}
                    if w0:
                        new0[x] = (new0[x] + w0 * mul) % MOD
                    if w1:
                        new1[x] = (new1[x] + w1 * mul) % MOD
            else:
                for x in range(S + 1):
                    w0 = dp0[x]
                    w1 = dp1[x]
                    if w0 == 0 and w1 == 0:
                        continue
                    
                    # Up move: x -> x + ai
                    nx = x + ai
                    if nx <= S:
                        mul = pow_p[nx]
                        if w0:
                            new0[nx] = (new0[nx] + w0 * mul) % MOD
                        if w1:
                            new1[nx] = (new1[nx] + w1 * mul) % MOD
                    
                    # Down move: x -> x - ai (only if x >= ai)
                    if x >= ai:
                        nx = x - ai
                        mul = pow_p[nx]
                        hit_zero = (nx == 0)
                        if w0:
                            if hit_zero:
                                new1[nx] = (new1[nx] + w0 * mul) % MOD
                            else:
                                new0[nx] = (new0[nx] + w0 * mul) % MOD
                        if w1:
                            new1[nx] = (new1[nx] + w1 * mul) % MOD
            
            dp0, dp1 = new0, new1
        
        # Sum over all x of dp1 (paths that hit zero at some point)
        fp = sum(dp1) % MOD
        ans = (ans * fp) % MOD
    
    return ans

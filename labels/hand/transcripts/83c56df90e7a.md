def solve_grid(n: int, r: str, c: str) -> Optional[List[str]]:
    """ Given an integer N and strings R and C of length N consisting of 'A', 'B', and 'C',
    determine if it's possible to fill an N×N grid such that:
    - Each row and column contains exactly one 'A', one 'B', and one 'C'
    - The leftmost character written in the i-th row matches the i-th character of R
    - The topmost character written in the i-th column matches the i-th character of C
    
    Empty cells are represented by '.' in the output.
    
    Args:
        n: The size of the grid (3 <= n <= 5)
        r: String of length n specifying leftmost characters in each row
        c: String of length n specifying topmost characters in each column
    
    Returns:
        A list of n strings representing the filled grid if possible, None otherwise.
        Each string has length n, where '.' represents empty cells.
    
    >>> solve_grid(3, 'ABC', 'ABC')
    ['ABC', 'BCA', 'CAB']
    >>> solve_grid(3, 'AAA', 'BBB')
    None
    """
    # Precompute all valid row configurations for each row
    # A row configuration is a tuple (pos_A, pos_B, pos_C)
    chars = ['A', 'B', 'C']
    
    def generate_row_configs(row_idx):
        """Generate all (pos_A, pos_B, pos_C) for the given row that satisfy leftmost constraint."""
        configs = []
        for positions in permutations(range(n), 3):
            pos_A, pos_B, pos_C = positions
            # Find leftmost (minimum column index)
            min_pos = min(positions)
            if min_pos == pos_A:
                leftmost_char = 'A'
            elif min_pos == pos_B:
                leftmost_char = 'B'
            else:
                leftmost_char = 'C'
            if leftmost_char == r[row_idx]:
                configs.append((pos_A, pos_B, pos_C))
        return configs
    
    all_configs = [generate_row_configs(i) for i in range(n)]
    
    # Backtracking state
    col_a = [False] * n  # whether column j already has an A
    col_b = [False] * n
    col_c = [False] * n
    col_topmost = [None] * n  # topmost character in column j, None if not yet set
    row_assignments = []  # list of (pos_A, pos_B, pos_C) for assigned rows
    
    def backtrack(row_idx):
        if row_idx == n:
            return True
        
        for config in all_configs[row_idx]:
            pos_A, pos_B, pos_C = config
            
            # Check column constraints
            ok = True
            # Check A
            if col_a[pos_A]:
                ok = False
            elif col_topmost[pos_A] is None and 'A' != c[pos_A]:
                ok = False
            # Check B
            if ok and col_b[pos_B]:
                ok = False
            elif ok and col_topmost[pos_B] is None and 'B' != c[pos_B]:
                ok = False
            # Check C
            if ok and col_c[pos_C]:
                ok = False
            elif ok and col_topmost[pos_C] is None and 'C' != c[pos_C]:
                ok = False
            
            if not ok:
                continue
            
            # Apply
            col_a[pos_A] = True
            col_b[pos_B] = True
            col_c[pos_C] = True
            old_topmost = [col_topmost[pos_A], col_topmost[pos_B], col_topmost[pos_C]]
            if col_topmost[pos_A] is None:
                col_topmost[pos_A] = 'A'
            if col_topmost[pos_B] is None:
                col_topmost[pos_B] = 'B'
            if col_topmost[pos_C] is None:
                col_topmost[pos_C] = 'C'
            row_assignments.append(config)
            
            if backtrack(row_idx + 1):
                return True
            
            # Undo
            row_assignments.pop()
            col_topmost[pos_A], col_topmost[pos_B], col_topmost[pos_C] = old_topmost
            col_c[pos_C] = False
            col_b[pos_B] = False
            col_a[pos_A] = False
        
        return False
    
    if not backtrack(0):
        return None
    
    # Build grid from assignments
    grid = [['.'] * n for _ in range(n)]
    for i, (pos_A, pos_B, pos_C) in enumerate(row_assignments):
        grid[i][pos_A] = 'A'
        grid[i][pos_B] = 'B'
        grid[i][pos_C] = 'C'
    
    return [''.join(row) for row in grid]

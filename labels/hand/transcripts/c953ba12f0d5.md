def sum_concatenated_permutations(n: int) -> int:
    """ Given a positive integer n, calculate the sum of concatenated permutations modulo 998244353.
    
    For each permutation P of (1, 2, ..., n), concatenate all numbers in P to form a single integer.
    For example, if P = (1, 20, 34), the concatenated result is 12034.
    
    Return the sum of all such concatenated integers for all n! permutations, modulo 998244353.
    
    Args:
        n: A positive integer (1 <= n <= 2 * 10^5)
    
    Returns:
        The sum of concatenated permutations modulo 998244353
    
    >>> sum_concatenated_permutations(3)
    1332
    >>> sum_concatenated_permutations(1)
    1
    """
    MOD = 998244353

    # Step 1: group numbers 1..n by their digit length
    groups = {}  # digit_length -> [count, sum]
    for i in range(1, n + 1):
        d = len(str(i))
        if d not in groups:
            groups[d] = [0, 0]
        groups[d][0] += 1
        groups[d][1] += i

    d_list = sorted(groups.keys())
    c = [groups[d][0] for d in d_list]  # counts per digit length
    s = [groups[d][1] % MOD for d in d_list]  # sums per digit length
    v = [pow(10, d, MOD) for d in d_list]   # 10^d per digit length
    D = len(d_list)

    # Step 2: compute elementary symmetric sums e_k of the multiset {10^{len(i)} : i=1..n}
    # Using the recurrence derived from the exponential generating function:
    #   P(y) = prod_d (1 + v_d * y)^{c_d}
    #   log P(y) = sum_d c_d * log(1 + v_d y)
    #   k * e_k = sum_{m=1..k} (-1)^{m-1} * B_m * e_{k-m}  where B_m = sum_d c_d * v_d^m
    # Efficient O(n*D) via auxiliary sequences R_d[k] = sum_{t=0..k} (-1)^t * v_d^t * e_{k-t}
    e = [0] * (n + 1)
    e[0] = 1
    R = [1] * D  # R_d[0] = 1 for all d

    for k in range(1, n + 1):
        total = 0
        for d_idx in range(D):
            total = (total + c[d_idx] * v[d_idx] % MOD * R[d_idx]) % MOD
        e[k] = total * pow(k, MOD - 2, MOD) % MOD

        for d_idx in range(D):
            R[d_idx] = (e[k] - v[d_idx] * R[d_idx]) % MOD

    # Step 3: precompute factorials modulo MOD
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i - 1] * i % MOD

    # Step 4: for each digit-length group, compute
    #   W_d = sum_{k=0..n-1} k! * (n-1-k)! * e_k^{(-d)}
    # where e_k^{(-d)} is the k-th elementary symmetric sum after removing one element of group d.
    # Recurrence: e_0^{(-d)} = 1,  e_k^{(-d)} = e_k - v_d * e_{k-1}^{(-d)}
    ans = 0
    for d_idx in range(D):
        cur = 1  # e_0^{(-d)}
        W = fact[0] * fact[n - 1] % MOD * cur % MOD
        for k in range(1, n):
            cur = (e[k] - v[d_idx] * cur) % MOD
            W = (W + fact[k] * fact[n - 1 - k] % MOD * cur) % MOD
        ans = (ans + s[d_idx] * W) % MOD

    return ans

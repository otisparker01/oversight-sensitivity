from typing import List


def max_score(grid: List[List[int]]) -> int:
    """ Given a 2D matrix of positive integers, select one or more cells such that:
    - No two selected cells are in the same row
    - The values in the selected cells are unique
    
    Return the maximum sum of values you can achieve.
    
    Args:
        grid: A 2D list where 1 <= len(grid), len(grid[i]) <= 10 and 1 <= grid[i][j] <= 100
    
    >>> max_score([[1,2,3],[4,3,2],[1,1,1]])
    8
    >>> max_score([[8,7,6],[8,3,2]])
    15
    """
    R = len(grid)
    # Map each distinct value to the set of rows that contain it
    value_to_rows: dict[int, set] = {}
    for r, row in enumerate(grid):
        for v in row:
            value_to_rows.setdefault(v, set()).add(r)
    
    # DP over bitmask of used rows
    size = 1 << R
    dp = [-1] * size
    dp[0] = 0    
    for v, rows in value_to_rows.items():
        new_dp = dp[:]  # copy for this value iteration
        for mask in range(size):
            cur = dp[mask]
            if cur >= 0:
                for r in rows:
                    if not (mask & (1 << r)):
                        new_mask = mask | (1 << r)
                        cand = cur + v
                        if cand > new_dp[new_mask]:
                            new_dp[new_mask] = cand
        dp = new_dp
    
    return max(dp)

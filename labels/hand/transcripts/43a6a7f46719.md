```python
from typing import List


def min_cost_to_sort(n: int, permutation: List[int]) -> int:
    """ Given a permutation of integers from 1 to n, find the minimum total cost to sort it in 
    ascending order. You can repeatedly swap adjacent elements, where swapping elements at 
    positions i and i+1 (1-indexed) costs i.
    
    Args:
        n: The length of the permutation
        permutation: A list of integers from 1 to n in some order
    
    Returns:
        The minimum total cost required to sort the permutation in ascending order
    
    >>> min_cost_to_sort(3, [3, 2, 1])
    4
    >>> min_cost_to_sort(5, [2, 4, 1, 3, 5])
    6
    >>> min_cost_to_sort(2, [1, 2])
    0
    """
    # Work on a mutable copy of the permutation.
    arr = list(permutation)
    total_cost = 0
    
    # Process elements from largest to smallest.
    # For each element, move it directly to its target position (val - 1 in 0-indexed).
    for val in range(n, 0, -1):
        # Find the current index of val.
        pos = arr.index(val)       # 0-indexed
        target = val - 1           # 0-indexed target position
        
        # Move val rightward if it is left of its target.
        while pos < target:
            # Swap with the element at pos+1 (right neighbour).
            # Cost is the 1-indexed position of the left element = pos + 1.
            total_cost += pos + 1
            arr[pos], arr[pos + 1] = arr[pos + 1], arr[pos]
            pos += 1
        
        # Move val leftward if it is right of its target.
        while pos > target:
            # Swap with the element at pos-1 (left neighbour).
            # Cost is the 1-indexed position of the left element = pos.
            total_cost += pos
            arr[pos], arr[pos - 1] = arr[pos - 1], arr[pos]
            pos -= 1
    
    return total_cost
```

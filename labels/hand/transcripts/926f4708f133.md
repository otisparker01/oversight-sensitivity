def _count_set_bits(m: int, p: int) -> int:
    """Count numbers in [1, m] with bit p set."""
    if m <= 0:
        return 0
    cycle = 1 << (p + 1)
    full_cycles = (m + 1) // cycle
    remainder = (m + 1) % cycle
    half_cycle = 1 << p
    return full_cycles * half_cycle + max(0, remainder - half_cycle)


def _total_before(n: int) -> int:
    """Total number of elements contributed by integers 1 through n-1.
    This is sum_{k=1}^{n-1} popcount(k)."""
    if n <= 1:
        return 0
    m = n - 1
    total = 0
    p = 0
    while (1 << p) <= m:
        total += _count_set_bits(m, p)
        p += 1
    return total


def _sum_exponents_before(n: int) -> int:
    """Sum of exponents contributed by integers 1 through n-1."""
    if n <= 1:
        return 0
    m = n - 1
    total = 0
    p = 0
    while (1 << p) <= m:
        total += p * _count_set_bits(m, p)
        p += 1
    return total


def _find_integer_and_offset(N: int) -> tuple:
    """Given N (number of elements before the target), find which integer
    and which offset within its powerful array.
    Returns (n, offset) where:
        - total_before(n) <= N < total_before(n+1)
        - offset = N - total_before(n)
    """
    lo = 1
    hi = 10**15 + 1  # exclusive upper bound
    while lo < hi:
        mid = (lo + hi) //2
        if _total_before(mid) <= N:
            lo = mid + 1
        else:
            hi = mid
    n = lo - 1
    offset = N - _total_before(n)
    return n, offset


def _sum_first_offset_bits(n: int, offset: int) -> int:
    """Sum of exponents of the first `offset` set bits of integer n."""
    if offset <=0:

            cnt += 1

        p += 1

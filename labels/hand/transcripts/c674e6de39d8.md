def count_valid_grids(H: int, W: int, grid: list[str]) -> int:
    """ Given a grid with H rows and W columns consisting of '1', '2', '3', and '?',
    count how many ways we can replace each '?' with '1', '2', or '3' such that
    any two adjacent (edge-sharing) cells contain different digits.
    
    Return the count modulo 998244353.
    
    Args:
        H: Number of rows (1 <= H, H * W <= 200)
        W: Number of columns (1 <= W, H * W <= 200)
        grid: List of strings representing the grid, where grid[i] is the i-th row
        
    Returns:
        The number of valid grid configurations modulo 998244353
        
    >>> count_valid_grids(2, 2, ['1?', '??'])
    6
    >>> count_valid_grids(2, 3, ['123', '3?1'])
    0
    >>> count_valid_grids(1, 1, ['?'])
    3
    """
    MOD = 998244353
    
    # Check for any pre-existing conflict between adjacent fixed cells.
    # A grid that already violates the adjacency constraint cannot be
    # rescued by assigning values to '?' cells.
    for i in range(H):
        for j in range(W):
            ch = grid[i][j]
            if ch != '?':
                if j + 1 < W and grid[i][j + 1] != '?' and grid[i][j + 1] == ch:
                    return 0
                if i +1 < H and grid[i +1][j] != '?' and grid[i +1][j] == ch:
                    return 0
    
    # Transpose to ensure W <= H for smaller DP profile
    if H < W:
        new_grid = []
        for c in range(W):
            new_row = []
            for r in range(H):
                new_row.append(grid[r][c])
            new_grid.append(''.join(new_row))
        grid = new_grid
        H, W = W, H
    
    # DP state: dictionary mapping profile tuple to count
    # profile[i] is the color of the cell at the frontier column i
    # 0 means no cell (for initial row boundary)
    initial_profile = tuple([0] * W)
    dp = {initial_profile: 1}
    
    colors = ['1', '2', '3']
    
    for i in range(H):
        for j in range(W):
            new_dp = {}
            ch = grid[i][j]
            for profile, count in dp.items():
                # Determine allowed colors for this cell
                if ch == '?':
                    allowed = colors
                else:
                    allowed = [ch]
                
                top = profile[j]  # 0 if first row
                left = profile[j - 1] if j > 0 else '0'  # '0' sentinel, never conflicts
                
                for c in allowed:
                    if c == top:
                        continue
                    if c == left:
                        continue
                    # Build new profile
                    new_profile = list(profile)
                    new_profile[j] = c
                    new_profile = tuple(new_profile)
                    new_dp[new_profile] = (new_dp.get(new_profile, 0) + count) % MOD
            dp = new_dp
    
    # Sum all final states
    total = sum(dp.values()) % MOD
    return total

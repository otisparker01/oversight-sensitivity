    >>> sum_concatenated_permutations(1)
    1

        length <<= 1

        j = 0
        for i in range(1, n_len):
            bit = n_len >> 1
            while j & bit:
                j ^= bit
                bit >>= 1
            j ^= bit

                    a[i + k_] = (u + v) % MOD
                    a[i + k_ + half] = (u - v) % MOD

        ntt(A, True)

    total = 0

        D = g['D']

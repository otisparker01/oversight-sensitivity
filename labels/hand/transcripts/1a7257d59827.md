import array


def min_weight_hamiltonian_walk(n: int, m: int, edges: List[Tuple[int, int, int]]) -> Optional[int]:
    """ Given a weighted directed graph with n vertices (numbered 1 to n) and m edges,
    determine if there exists a walk that visits each vertex at least once.
    If such a walk exists, return the minimum total weight of edges traversed.
    If no such walk exists, return None.
    
    The graph is guaranteed not to contain negative cycles.
    Edges can be traversed multiple times, with their weight counted each time.
    
    Args:
        n: Number of vertices (2 <= n <=20)
        m: Number of edges (1 <= m <= n*(n-1))
        edges: List of tuples (u, v, w) representing directed edges from u to v with weight w
               where 1 <= u, v <= n, u != v, and -10^6 <= w <=10^6
    
    Returns:
        The minimum total weight if a valid walk exists, None otherwise.
    
    >>> min_weight_hamiltonian_walk(3, 4, [(1, 2,5), (2, 1, -3), (2, 3, -4), (3, 1, 100)])
    -2
    >>> min_weight_hamiltonian_walk(3,2, [(1, 2,0), (2, 1,0)])
    None
    >>> min_weight_hamiltonian_walk(2,1, [(2, 1,0)])
    0
    """
    INF = 10 ** 18
    
    # Initialize distance matrix
    dist = [[INF] * n for _ in range(n)]
    for i in range(n):
        dist[i][i] = 0
    for u, v, w in edges:
        u -= 1
        v -= 1
        if w < dist[u][v]:
            dist[u][v] = w
    
    # Floyd-Warshall
    for k in range(n):
        dk = dist[k]
        for i in range(n):
            dik = dist[i][k]
            if dik == INF:
                continue
            di = dist[i]
            for j in range(n):
                nd = dik + dk[j]
                if nd < di[j]:
                    di[j] = nd
    
    size = 1 << n
    full = size - 1
    
    # Precompute bit lookup
    bit_idx = [0] * (1 << n)
    for i in range(n):
        bit_idx[1 << i] = i
    
    # DP: dp[mask][last] as flat array for speed
    # Use array.array for memory efficiency
    total = size * n
    dp = array.array('q', [INF]) * total
    for i in range(n):
        dp[(1 << i) * n + i] = 0
    
    # Push DP: for each reachable state, push to larger masks
    for mask in range(size):
        base = mask * n
        m = mask
        while m:
            lsb = m & -m
            last = bit_idx[lsb]
            m ^= lsb
            
            cur = dp[base + last]
            if cur == INF:
                continue
            
            di = dist[last]
            not_m = full ^ mask
            n = not_m
            while n:
                nlsb = n & -n
                nxt = bit_idx[nlsb]
                n ^= nlsb
                
                nd = cur + di[nxt]
                idx = (mask | nlsb) * n + nxt
                if nd < dp[idx]:
                    dp[idx] = nd
    
    try:
        ans = min(dp[full * n: full * n + n])
    except ValueError:
        # Fallback: iterate manually
        ans = INF
        base = full * n
        for i in range(n):
            v = dp[base + i]
            if v < ans:
                ans = v
    return ans if ans < INF // 2 else None

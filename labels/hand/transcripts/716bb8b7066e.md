```python
from typing import List

MOD = 10 ** 9 + 7


def _count_min_gap_ge(nums_sorted: List[int], k: int, d: int) -> int:
    """Return the number of k-element subsequences of nums_sorted
    (already sorted) whose minimum adjacent difference is >= d."""
    n = len(nums_sorted)
    # dp[i][j] = number of j-element subsets from nums_sorted[0..i]
    #            with minimum adjacent gap >= d
    dp = [[0] * (k + 1) for _ in range(n)]

    for i in range(n):
        dp[i][0] = 1
        dp[i][1] = i + 1

    # Precompute prev[i]: largest index < i such that
    # nums_sorted[i] - nums_sorted[prev[i]] >= d
    prev = [-1] * n
    left = 0
    for i in range(n):
        while left < i and nums_sorted[i] - nums_sorted[left] >= d:
            left += 1
        prev[i] = left - 1  # left is the first index where diff >= d fails,
                            # so left-1 is the last index where it holds

    for j in range(2, k + 1):
        for i in range(j - 1, n):
            p = prev[i]
            if p >= 0:
                dp[i][j] = (dp[i - 1][j] + dp[p][j - 1]) % MOD
            else:
                dp[i][j] = dp[i - 1][j]

    return dp[n - 1][k]


def sum_of_powers(nums: List[int], k: int) -> int:
    """ Given an integer array nums and a positive integer k, return the sum of powers 
    of all subsequences of nums which have length equal to k.
    
    The power of a subsequence is defined as the minimum absolute difference between 
    any two elements in the subsequence.
    
    Return the result modulo 10^9 + 7.
    
    Arg:
        nums: List of integers where -10^8 <= nums[i] <=10^8
        k: Length of subsequences to consider (2 <= k <= len(nums))
    
    Returns:
        Sum of powers modulo 10^9 +7
    
    >>> sum_of_powers([1, 2, 3,4], 3)
    4
    >>> sum_of_powers([2,2], 2)
    0
    >>> sum_of_powers([4, 3, -1], 2)
    10
    """
    s = sorted(nums)
    n = len(s)

    # Collect all unique differences between any two elements
    diffs = set()
    for i in range(n):
        for j in range(i + 1, n):
            diffs.add(s[j] - s[i])
    diffs = sorted(diffs)

    # Compute count(>=d) for each distinct difference
    counts = {}
    for d in diffs:
        counts[d] = _count_min_gap_ge(s, k, d)

    ans = 0
    for idx, d in enumerate(diffs):
        if idx + 1 < len(diffs):
            next_d = diffs[idx + 1]
            cnt = (counts[d] - counts[next_d]) % MOD
        else:
            cnt = counts[d] % MOD
        ans = (ans + d * cnt) % MOD

    return ans
```

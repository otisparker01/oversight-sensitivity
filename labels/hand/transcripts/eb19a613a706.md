from functools import lru_cache


def tic_tac_toe_winner(grid: List[List[int]]) -> str:
    """ Play a modified tic-tac-toe game on a 3x3 grid where each cell has a value.
    
    Takahashi (first player) and Aoki (second player) take turns choosing white cells.
    When a player chooses a cell, they score the points in that cell and paint it their color
    (Takahashi: red, Aoki: blue).
    
    The game ends when:
    - A player gets 3 consecutive cells of their color (row/column/diagonal) - that player wins
    - All cells are painted - the player with higher total score wins
    
    Both players play optimally. The sum of all values in the grid is guaranteed to be odd.
    
    Args:
        grid: A 3x3 grid of integers representing cell values
        
    Returns:
        "Takahashi" if Takahashi wins, "Aoki" if Aoki wins
        
    >>> tic_tac_toe_winner([[0, 0, 0], [0, 1, 0], [0, 0, 0]])
    'Takahashi'
    >>> tic_tac_toe_winner([[-1, 1, 0], [-4, -2, -5], [-4, -1, -5]])
    'Aoki'
    """
    values = tuple(grid[0] + grid[1] + grid[2])
    
    # All 8 winning lines
    lines = [
        (0, 1, 2), (3, 4, 5), (6, 7, 8),  # rows
        (0, 3, 6), (1, 4, 7), (2, 5, 8),  # columns
        (0, 4, 8), (2, 4, 6)              # diagonals
    ]
    
    @lru_cache(maxsize=None)
    def minimax(board: tuple, player: int) -> bool:
        """Returns True if Takahashi wins, False if Aoki wins.
        player: 1 = Takahashi, 2 = Aoki."""
        # Check if either player has a winning line
        for a, b, c in lines:
            if board[a] == board[b] == board[c] == 1:
                return True
            if board[a] == board[b] == board[c] == 2:
                return False
        
        # Check if board is full (score comparison)
        if 0 not in board:
            takahashi_score = sum(values[i] for i in range(9) if board[i] == 1)
            aoki_score = sum(values[i] for i in range(9) if board[i] == 2)
            return takahashi_score > aoki_score
        
        # Try each move
        if player == 1:  # Takahashi's turn (maximizer)
            for i in range(9):
                if board[i] == 0:
                    new_board = list(board)
                    new_board[i] = 1
                    if minimax(tuple(new_board), 2):
                        return True
            return False
        else:  # Aoki's turn (minimizer)
            for i in range(9):
                if board[i] == 0:
                    new_board = list(board)
                    new_board[i] = 2
                    if not minimax(tuple(new_board), 1):
                        return False
            return True
    
    result = minimax((0,) * 9, 1)
    return "Takahashi" if result else "Aoki"

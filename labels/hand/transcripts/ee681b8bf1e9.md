MOD = 998244353


def count_strings_by_lcs_length(n: int, m: int, s: str) -> List[int]:
    """ Given a lowercase English string s of length n and an integer m, 
    for each k from 0 to n, count the number of lowercase English strings 
    of length m whose longest common subsequence with s has length exactly k.
    
    Return a list of n+1 integers where the i-th element is the count 
    (modulo 998244353) for k=i.
    
    Args:
        n: Length of string s (1 <= n <= 10)
        m: Length of strings to count (1 <= m <= 100)
        s: A lowercase English string of length n
        
    Returns:
        A list of n+1 integers representing counts modulo 998244353
    
    >>> count_strings_by_lcs_length(2, 2, 'ab')
    [576, 99, 1]
    >>> count_strings_by_lcs_length(3, 4, 'aaa')
    [390625, 62500, 3750, 101]
    """
    size = 1 << n  # number of states (2^n)
    
    # Precompute transitions: trans[mask][c] = new_mask
    trans = [[0] * 26 for _ in range(size)]
    
    for mask in range(size):
        # Decode mask to DP column values
        old_dp = [0] * (n + 1)
        for i in range(1, n + 1):
            old_dp[i] = old_dp[i - 1] + ((mask >> (i - 1)) & 1)
        
        for c_idx in range(26):
            ch = chr(ord('a') + c_idx)
            new_dp = [0] * (n + 1)
            for i in range(1, n + 1):
                if s[i - 1] == ch:
                    new_dp[i] = old_dp[i - 1] + 1
                else:
                    new_dp[i] = max(old_dp[i], new_dp[i - 1])
            # Encode new column to mask
            new_mask = 0
            for i in range(1, n + 1):
                if new_dp[i] > new_dp[i - 1]:
                    new_mask |= 1 << (i - 1)
            trans[mask][c_idx] = new_mask
    
    # DP over lengths of t
    dp = [0] * size
    dp[0] = 1  # empty string
    
    for _ in range(m):
        ndp = [0] * size
        for mask in range(size):
            val = dp[mask]
            if val == 0:
                continue
            for c_idx in range(26):
                new_mask = trans[mask][c_idx]
                ndp[new_mask] = (ndp[new_mask] + val) % MOD
        dp = ndp
    
    # Aggregate by LCS length (popcount of mask)
    result = [0] * (n + 1)
    for mask in range(size):
        k = mask.bit_count()
        result[k] = (result[k] + dp[mask]) % MOD
    
    return result

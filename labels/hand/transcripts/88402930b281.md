```python
def min_changes_for_semi_palindromes(s: str, k: int) -> int:
    """ Given a string s and an integer k, partition s into k substrings such that the sum 
    of the number of letter changes required to turn each substring into a semi-palindrome 
    is minimized. Return the minimum number of letter changes required.
    
    """
    n = len(s)

    # Precompute cost[i][j] = min changes to make s[i:j] a semi-palindrome
    INF = 10 ** 9
    cost = [[INF] * (n + 1) for _ in range(n)]

    for i in range(n):
        for j in range(i + 2, n + 1):
            length = j - i
            best = INF
            for d in range(1, length):
                if length % d != 0:
                    continue
                changes = 0
                for r in range(d):
                    left = i + r
                    right = j - d + r
                    while left < right:
                        if s[left] != s[right]:
                            changes += 1
                        left += d
                        right -= d
                if changes < best:
                    best = changes
            cost[i][j] = best

    dp = [[INF] * (k + 1) for _ in range(n + 1)]
    dp[0][0] = 0

    for p in range(1, n + 1):
        max_t = min(k, p // 2)
        for t in range(1, max_t + 1):
            best = INF
            for prev in range(2 * (t - 1), p - 1):
                if cost[prev][p] != INF and dp[prev][t - 1] != INF:
                    cand = dp[prev][t - 1] + cost[prev][p]
                    if cand < best:
                        best = cand
            dp[p][t] = best

    return dp[n][k]
```

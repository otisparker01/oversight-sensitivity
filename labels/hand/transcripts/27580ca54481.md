```python
def find_largest_400_number(a: int) -> int:
    """ Find the largest 400 number not exceeding a.
    
    A positive integer N is a 400 number if and only if:
    1. N has exactly 2 distinct prime factors.
    2. For each prime factor p of N, p divides N an even number of times.
       (i.e., the maximum k such that p^k divides N is even)
    
    Args:
        a: An integer where 36 <= a <= 10^12
    
    Returns:
        The largest 400 number not exceeding a
    
    >>> find_largest_400_number(404)
    400
    >>> find_largest_400_number(36)
    36
    >>> find_largest_400_number(60)
    36
    >>> find_largest_400_number(1000000000000)
    1000000000000
    >>> find_largest_400_number(123456789)
    123454321
    """
    # A 400 number N = p^(2i) * q^(2j) = (p^i * q^j)^2
    # So N is the square of m = p^i * q^j, where m has exactly 2 distinct prime factors.
    # We iterate m downwards from floor(sqrt(a)) and find the first m with exactly 2 distinct primes.
    
    m_max = math.isqrt(a)
    
    # Sieve to count distinct prime factors for all numbers up to m_max
    dpf = [0] * (m_max + 1)
    for i in range(2, m_max + 1):
        if dpf[i] == 0:  # i is prime
            for j in range(i, m_max + 1, i):
                dpf[j] += 1
    
    for m in range(m_max, 5, -1):  # smallest m with 2 distinct primes is 6
        if dpf[m] == 2:
            return m * m
    
    # Should never reach here given a >= 36
    return 36
```

from typing import List, Tuple
import bisect


def can_reach_end(n: int, m: int, a: int, b: int, bad_intervals: List[Tuple[int, int]]) -> bool:
    """ Determine if it's possible to move from square 1 to square N by repeatedly jumping.
    
    There are N squares arranged in a row, labeled 1 to N. You start at square 1 and want to reach square N.
    In each move, if you're at square x, you can jump to square x + i where:
    - a <= i <= b
    - x + i <= n
    - square x + i is not in any of the bad intervals
    
    A square j is bad if there exists some interval (l_i, r_i) such that l_i <= j <= r_i.
    
    Args:
        n: The number of squares (2 <= n <= 10^12)
        m: The number of bad intervals (0 <= m <= 2*10^4)
        a: The minimum jump size (1 <= a <= b <= 20)
        b: The maximum jump size
        bad_intervals: List of tuples (l_i, r_i) representing bad intervals.
                      The intervals are sorted and non-overlapping.
    
    Returns:
        True if it's possible to reach square n from square 1, False otherwise.
    
    >>> can_reach_end(24, 2, 3, 5, [(7, 8), (17, 20)])
    True
    >>> can_reach_end(30, 1, 5, 8, [(4, 24)])
    False
    >>> can_reach_end(100, 4, 10, 11, [(16, 18), (39, 42), (50, 55), (93, 99)])
    True
    """
    # Build good intervals (complement of bad intervals)
    good_intervals = []
    prev_end = 1
    for l, r in bad_intervals:
        if prev_end < l:
            good_intervals.append((prev_end, l - 1))
        prev_end = r + 1
    if prev_end <= n:
        good_intervals.append((prev_end, n))
    
    # If square 1 or square n is bad, impossible
    if not good_intervals or good_intervals[0][0] != 1:
        return False
    if good_intervals[-1][1] != n:
        return False
    
    k = len(good_intervals)
    # entry_sets[i] = set of positions in good_intervals[i] that are directly reachable
    # from previous segments' exit frontiers
    entry_sets = [set() for _ in range(k)]
    # exit_frontiers[i] = set of reachable positions in the last b positions of segment i
    exit_frontiers = [None] * k
    
    # Initially, position 1 is reachable (it's in the first segment)
    if good_intervals[0][0] == 1:
        entry_sets[0].add(1)
    
    # Helper: find which good interval contains position p
    good_starts = [L for L, R in good_intervals]
    
    def find_segment(pos):
        # Returns index of good interval containing pos, or -1 if none
        idx = bisect.bisect_right(good_starts, pos) - 1
        if idx >= 0 and good_intervals[idx][0] <= pos <= good_intervals[idx][1]:
            return idx
        return -1
    
    for i in range(k):
        if not entry_sets[i]:
            continue
        
        L, R = good_intervals[i]
        entry = entry_sets[i]
        
        # Simulate within this segment to compute exit frontier
        seg_len = R - L + 1
        limit = min(seg_len, a * b + b)
        
        reachable = [False] * (limit + b)
        for pos in entry:
            offset = pos - L
            if offset < limit:
                reachable[offset + b] = True
        
        for idx in range(b, limit + b):
            if reachable[idx]:
                continue
            for j in range(a, b + 1):
                if reachable[idx - j]:
                    reachable[idx] = True
                    break
        
        if limit == seg_len:
            # Fully simulated
            exit_frontier = set()
            for pos in range(max(L, R - b + 1), R + 1):
                offset = pos - L
                if reachable[offset + b]:
                    exit_frontier.add(pos)
        else:
            # Long segment
            if a == b:
                residues = set()
                for idx in range(b, limit + b):
                    if reachable[idx]:
                        residues.add((L + idx - b) % a)
                exit_frontier = set()
                for pos in range(max(L, R - b + 1), R + 1):
                    if pos % a in residues:
                        exit_frontier.add(pos)
            else:
                saturated = False
                for idx in range(b, limit + b - a + 1):
                    if all(reachable[idx + k] for k in range(a)):
                        saturated = True
                        break
                if saturated:
                    exit_frontier = set(range(max(L, R - b + 1), R + 1))
                else:
                    exit_frontier = set()
                    for pos in range(max(L, R - b + 1), R + 1):
                        offset = pos - L
                        if offset < limit and reachable[offset + b]:
                            exit_frontier.add(pos)
        
        exit_frontiers[i] = exit_frontier
        
        # Distribute to later segments
        for x in exit_frontier:
            for j in range(a, b + 1):
                y = x + j
                if y > n:
                    continue
                seg_idx = find_segment(y)
                if seg_idx > i:
                    entry_sets[seg_idx].add(y)
    
    # Check if n is reachable (it's in the last segment)
    if exit_frontiers[-1] is not None:
        return n in exit_frontiers[-1]
    return False

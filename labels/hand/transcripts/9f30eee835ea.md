import math

def count_power_representations(n: int) -> int:
    """ Count how many integers x between 1 and n (inclusive) can be expressed as x = a^b
    where a is a positive integer and b is an integer greater than or equal to 2.
    
    Args:
        n: The upper bound (1 <= n <= 10^18)
    
    Returns:
        The count of integers that can be expressed as a^b where b >= 2
    
    >>> count_power_representations(99)
    12
    >>> count_power_representations(1000000000000000000)
    1001003332
    """
    # Count perfect squares (includes 1 = 1^2)
    squares = math.isqrt(n)
    
    # Count higher powers (b >= 3, odd only since even b produce squares)
    higher = set()
    
    max_b = n.bit_length()  # floor(log2(n))
    
    for b in range(3, max_b + 1):
        if b % 2 == 0:
            continue  # even exponents produce only squares
        if (1 << b) > n:
            break  # 2^b > n, no more values
        
        # Integer k-th root: floor(n^(1/b))
        lo, hi = 1, int(n ** (1.0 / b)) + 2
        while lo <= hi:
            mid = (lo + hi) // 2
            if mid ** b <= n:
                lo = mid + 1
            else:
                hi = mid - 1
        max_a = hi
        
        for a in range(2, max_a + 1):
            x = a ** b
            # Skip if x is a perfect square (already counted)
            if math.isqrt(x) ** 2 != x:
                higher.add(x)
    
    return squares + len(higher)

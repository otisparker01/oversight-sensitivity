```python
from math import factorial
from collections import Counter


def count_k_palindromic_numbers(n: int, k: int) -> int:
    """ Count the number of good integers containing exactly n digits.
    
    An integer x is called k-palindromic if:
    - x is a palindrome (reads the same forwards and backwards)
    - x is divisible by k
    
    An integer is called good if its digits can be rearranged to form a k-palindromic integer.
    
    Note: Integers must not have leading zeros, neither before nor after rearrangement.
    
    Args:
        n: The number of digits (1 <= n <= 10)
        k: The divisor for k-palindromic property (1 <= k <= 9)
    
    Returns:
        The count of good integers with exactly n digits
    
    >>> count_k_palindromic_numbers(3, 5)
    27
    >>> count_k_palindromic_numbers(1, 4)
    2
    >>> count_k_palindromic_numbers(5, 6)
    2468
    """
    # Step 1: Generate all n-digit palindromes (no leading zero)
    # A palindrome is determined by its first half (including middle for odd n)
    half_len = (n + 1) // 2  # ceil(n/2)
    
    # Precompute factorials for permutation counting
    facts = [factorial(i) for i in range(n + 1)]
    
    # Collect unique digit multisets from valid k-palindromic numbers
    seen_signatures = set()
    
    # Generate all possible first halves
    # The first digit of the palindrome cannot be 0 (for n > 1)
    start = 10**(half_len - 1) if n > 1 else 0
    end = 10**half_len
    
    for first_half in range(start, end):
        s = str(first_half).zfill(half_len)
        # Build the full palindrome
        if n % 2 == 0:
            pal_str = s + s[::-1]
        else:
            pal_str = s + s[-2::-1]  # drop the middle digit for mirroring
        
        # Check divisibility
        if int(pal_str) % k != 0:
            continue
        
        # Count digits
        cnt = Counter(pal_str)
        signature = tuple(cnt.get(str(d), 0) for d in range(10))
        
        if signature in seen_signatures:
            continue
        seen_signatures.add(signature)
    
    # Step 2: For each unique multiset, count n-digit numbers (no leading zero)
    total = 0
    for sig in seen_signatures:
        # sig is a tuple of 10 counts
        # Total permutations = n! / prod(c_i!)
        denominator = 1
        for c in sig:
            denominator *= facts[c]
        total_perms = facts[n] // denominator
        
        # Subtract permutations with leading zero
        if sig[0] > 0:
            # Fix one zero at the front, permute remaining n-1 digits
            denom_zero = facts[sig[0] - 1]
            for i in range(1, 10):
                denom_zero *= facts[sig[i]]
            total_perms -= facts[n - 1] // denom_zero
        
        total += total_perms
    
    return total
```

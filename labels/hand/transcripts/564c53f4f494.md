    For each permutation P of (1, 2, ..., n), concatenate all numbers in P to form a single integer.
    For example, if P = (1, 20, 34), the concatenated result is 12034.
    
    Return the sum of all such concatenated integers for all n! permutations, modulo 998244353.
    
    Args:

    1

    groups = []

        j = 0
        for i in range(1, n):

            while j & bit:
                j ^= bit
                bit >>= 1
            j ^= bit
            if i < j:
                arr[i], arr[j] = arr[j], arr[i]

        while length <= n:

            if invert:
                wlen = pow(wlen, MOD-2, MOD)

                half = length // 2
                for j in range(i, i + half):

                    v = arr[j + half] * w % MOD
                    arr[j] = (u + v) % MOD
                    arr[j + half] = (u - v) % MOD
                    w = w * wlen % MOD

            n_inv = pow(n, MOD-2, MOD)
            for i in range(n):

    while size < len(a) + len(b) - 1:
        size <<= 1
    fa = a + [0] * (size - len(a))

def ntt(a, invert):
    n = len(a)
    j = 0
    for i in range(1, n):
        bit = n >> 1
        while j & bit:
            j ^= bit
            bit >>= 1
        j ^= bit
        if i < j:
            a[i], a[j] = a[j], a[i]
    
    length = 2

                v = a[j + length//2] * w % MOD
                a[j] = (u + v) % MOD

                w = w * wlen % MOD

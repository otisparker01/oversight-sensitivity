def calculate_black_area_doubled(A: int, B: int, C: int, D: int) -> int:
    """ Calculate twice the area of black-painted regions inside a rectangle on AtCoder's wallpaper pattern.
    
    The wallpaper pattern is defined as follows:
    - The plane is divided by lines: x = n (integer n), y = n (even n), and x + y = n (even n)
    - Adjacent regions along these lines have different colors (black or white)
    - The region containing (0.5, 0.5) is black
    
    Given a rectangle with bottom-left vertex at (A, B) and top-right vertex at (C, D),
    calculate the area of black regions inside it and return twice that area.
    
    Args:
        A, B: Coordinates of the bottom-left vertex
        C, D: Coordinates of the top-right vertex
        
    Constraints:
        -10^9 <= A, B, C, D <= 10^9
        A < C and B < D
        
    Returns:
        Twice the area of black regions (guaranteed to be an integer)
    
    >>> calculate_black_area_doubled(0, 0, 3, 3)
    10
    >>> calculate_black_area_doubled(-1, -2, 1, 3)
    11
    >>> calculate_black_area_doubled(-1000000000, -1000000000, 1000000000, 1000000000)
    4000000000000000000
    """
    import math
    
    def color(x, y):
        """Return 0 for black, 1 for white."""
        return (math.floor(x) + math.floor(y / 2) + math.floor((x + y) / 2)) & 1
    
    # Compute prefix function F(x, y) = twice black area in [0, x] x [0, y]
    # where if x < 0, we interpret [0, x] as [x, 0]
    
    def F(x, y):
        if x == 0 or y == 0:
            return 0
        
        x1, x2 = (0, x) if x > 0 else (x, 0)
        y1, y2 = (0, y) if y > 0 else (y, 0)
        
        return G(x1, y1, x2, y2)
    
    def G(x1, y1, x2, y2):
        """Twice black area in [x1, x2] x [y1, y2], all non-negative."""
        w = x2 - x1
        h = y2 - y1
        
        # Full 4x4 blocks
        full_x = w // 4
        full_y = h // 4
        
        result = full_x * full_y * 16  # each 4x4 block contributes 16
        
        rem_w = w % 4
        rem_h = h % 4
        
        # Right strip
        if rem_w > 0:
            x_start = x1 + full_x * 4
            result += rect_area(x_start, y1, x_start + rem_w, y1 + full_y * 4)
        
        # Top strip
        if rem_h > 0:
            y_start = y1 + full_y * 4
            result += rect_area(x1, y_start, x1 + full_x * 4, y_start + rem_h)
        
        # Top-right corner
        if rem_w > 0 and rem_h > 0:
            x_start = x1 + full_x * 4
            y_start = y1 + full_y * 4
            result += rect_area(x_start, y_start, x_start + rem_w, y_start + rem_h)
        
        return result
    
    def rect_area(x1, y1, x2, y2):
        """Twice black area in small rectangle. x2-x1 < 4, y2-y1 < 4."""
        w = x2 - x1
        h = y2 - y1
        N = 200  # samples per unit
        nx = max(1, int(N * w))
        ny = max(1, int(N * h))
        dx = w / nx
        dy = h / ny
        black = 0
        for i in range(nx):
            x = x1 + (i + 0.5) * dx
            for j in range(ny):
                y = y1 + (j + 0.5) * dy
                if color(x, y) == 0:
                    black += 1
        total = nx * ny
        # Result should be integer; use rounding
        return (2 * w * h * black + total // 2) // total
    
    return F(C, D) - F(A, D) - F(C, B) + F(A, B)

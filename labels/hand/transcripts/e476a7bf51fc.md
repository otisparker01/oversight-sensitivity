from typing import List


def max_1122_subarray_length(n: int, a: List[int]) -> int:
    """ Given a list of positive integers, find the maximum length of a contiguous subarray 
    that forms a 1122 sequence.
    
    A 1122 sequence is defined as a sequence that:
    1. Has even length
    2. For each i from 1 to len/2, elements at positions 2i-1 and 2i are equal (1-indexed)
    3. Each positive integer appears either 0 times or exactly 2 times in the sequence
    
    Args:
        n: The length of the array (1 <= n <= 2*10^5)
        a: The array of positive integers (1 <= a[i] <= n)
    
    Returns:
        The maximum length of a contiguous subarray that is a 1122 sequence
    
    >>> max_1122_subarray_length(8, [2, 3, 1, 1, 2, 2, 1, 1])
    4
    >>> max_1122_subarray_length(3, [1, 2, 2])
    2
    >>> max_1122_subarray_length(1, [1])
    0
    """
    def solve_chain(start: int) -> int:
        """
        Process subarrays whose pairs align with a given parity.
        
        For a given start index s (0 or 1), valid subarrays must have
        pairs at positions (s, s+1), (s+2, s+3), etc.
        Extract the values of good pairs (where the two elements are equal),
        then find the longest segment with all distinct values.
        """
        max_len = 0
        seen = set()
        left = 0  # index within the values list
        
        # Build the list of pair values (None for mismatched positions)
        vals = []
        pos = start
        while pos + 1 < n:
            if a[pos] == a[pos + 1]:
                vals.append(a[pos])
            else:
                vals.append(None)
            pos += 2
        
        # Sliding window over vals: find longest contiguous segment
        # with all distinct values (skipping None entries).
        for right, val in enumerate(vals):
            if val is None:
                seen.clear()
                left = right + 1
            elif val in seen:
                while vals[left] != val:
                    seen.discard(vals[left])
                    left += 1
                seen.discard(vals[left])
                left += 1
                seen.add(val)
            else:
                seen.add(val)
            
            max_len = max(max_len, 2 * (right - left + 1))
        
        return max_len
    
    return max(solve_chain(0), solve_chain(1))

from typing import List


def kth_alternating_permutation(n: int, k: int) -> List[int]:
    """ Given two integers n and k, return the k-th alternating permutation of the first n positive integers.
    
    An alternating permutation is a permutation where no two adjacent elements are both odd or both even.
    The permutations are sorted in lexicographical order.
    
    If there are fewer than k valid alternating permutations, return an empty list.
    
    Args:
        n: The number of elements in the permutation (1 <= n <= 100)
        k: The 1-indexed position of the desired permutation (1 <= k <= 10^15)
    
    Returns:
        The k-th alternating permutation, or an empty list if k is out of range
    
    >>> kth_alternating_permutation(4, 6)
    [3, 4, 1, 2]
    >>> kth_alternating_permutation(3, 2)
    [3, 2, 1]
    >>> kth_alternating_permutation(2, 3)
    []
    """
    # Precompute factorials
    fact = [1] * (n + 1)
    for i in range(1, n + 1):
        fact[i] = fact[i-1] * i
    
    def count_ways(odds: int, evens: int, need_odd: bool) -> int:
        """Number of ways to arrange remaining odds and evens starting with given parity."""
        if odds == 0 and evens == 0:
            return 1
        if need_odd:
            # Need odd first: possible if odds == evens or odds == evens + 1
            if odds == evens or odds == evens + 1:
                return fact[odds] * fact[evens]
            else:
                return 0
        else:
            # Need even first: possible if evens == odds or evens == odds + 1
            if evens == odds or evens == odds + 1:
                return fact[odds] * fact[evens]
            else:
                return 0
    
    # Build lists of odds and evens
    odds = [i for i in range(1, n + 1) if i % 2 == 1]
    evens = [i for i in range(1, n + 1) if i % 2 == 0]
    
    # Check total count
    total = 0
    odd_count = len(odds)
    even_count = len(evens)
    # Odd-first count
    if odd_count == even_count or odd_count == even_count + 1:
        total += fact[odd_count] * fact[even_count]
    # Even-first count
    if even_count == odd_count or even_count == odd_count + 1:
        total += fact[odd_count] * fact[even_count]
    
    if k > total:
        return []
    
    result = []
    # At each step, we need opposite parity of last placed element
    # For the first position, we can try both parities
    for pos in range(n):
        # Determine which parity we need at this position
        if pos == 0:
            # First position: try all numbers in increasing order
            candidates = sorted(odds + evens)
        else:
            last = result[-1]
            if last % 2 == 1:
                # Last was odd, need even
                candidates = evens
            else:
                # Last was even, need odd
                candidates = odds
        
        chosen = None
        for x in candidates:
            # Try this candidate
            rem_odds = len(odds)
            rem_evens = len(evens)
            if x % 2 == 1:
                rem_odds -= 1
                next_need_odd = False  # after odd, need even
            else:
                rem_evens -= 1
                next_need_odd = True   # after even, need odd
            
            cnt = count_ways(rem_odds, rem_evens, next_need_odd)
            if k <= cnt:
                chosen = x
                break
            else:
                k -= cnt
        
        if chosen is None:
            # Should not happen if k is valid
            return []
        
        result.append(chosen)
        if chosen % 2 == 1:
            odds.remove(chosen)
        else:
            evens.remove(chosen)
    
    return result

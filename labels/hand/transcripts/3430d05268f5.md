from typing import List

MOD = 10**9 + 7


def count_valid_permutations(n: int, requirements: List[List[int]]) -> int:
    """ Count the number of permutations of [0, 1, 2, ..., n-1] such that for each requirement
    [end_i, cnt_i], the prefix perm[0:end_i+1] has exactly cnt_i inversions.
    
    An inversion is a pair of indices (i, j) where i < j and perm[i] > perm[j].
    
    Return the count modulo 10^9 + 7.
    
    Args:
        n: The size of the permutation (2 <= n <= 300)
        requirements: List of [end_index, inversion_count] pairs where:
            - 0 <= end_index <= n-1
            - 0 <= inversion_count <= 400
            - All end_indices are unique
            - At least one requirement has end_index == n-1
    
    Returns:
        The number of valid permutations modulo 10^9 + 7
    
    >>> count_valid_permutations(3, [[2, 2], [0, 0]])
    2
    >>> count_valid_permutations(3, [[2, 2], [1, 1], [0, 0]])
    1
    >>> count_valid_permutations(2, [[0, 0], [1, 0]])
    1
    """
    # Parse requirements into a dict: end_index -> required inversion count
    req = {}
    max_inv = 0
    for end_i, cnt_i in requirements:
        req[end_i] = cnt_i
        if cnt_i > max_inv:
            max_inv = cnt_i
    
    # Quick sanity: for any prefix of length k = end_i + 1, 
    # the maximum possible inversions is k*(k-1)//2.
    # If any requirement exceeds this, answer is 0.
    for end_i, cnt_i in requirements:
        max_possible = (end_i + 1) * end_i // 2
        if cnt_i > max_possible:
            return 0
    
    # Also check monotonicity: inversions cannot decrease as prefix grows.
    # Sort requirements by end_index and verify cnt is non-decreasing.
    # (This is an optimization but also catches trivial impossibilities.)
    
    # DP: dp[i][j] = number of ways for prefix of length i with j inversions.
    # We only need to track up to max_inv inversions (since requirements cap there).
    max_j = max_inv
    
    # dp for current length
    dp = [0] * (max_j + 1)
    dp[0] = 1  # empty prefix has 0 inversions
    
    for i in range(1, n + 1):
        # i is the new prefix length (we are adding element i-1)
        ndp = [0] * (max_j + 1)
        
        # Compute prefix sums of dp for efficient sliding window
        # dp[j] = sum_{d=0}^{min(i-1, j)} prev_dp[j-d]
        # We can use a sliding window sum.
        window_sum = 0
        for j in range(max_j + 1):
            # Add dp[j] to the window
            window_sum = (window_sum + dp[j]) % MOD
            # If window extends beyond i-1, remove the leftmost element
            if j - (i - 1) - 1 >= 0:
                window_sum = (window_sum - dp[j - (i - 1) - 1]) % MOD
            ndp[j] = window_sum
        
        # Apply requirement for end_index = i-1 (prefix length = i)
        if (i - 1) in req:
            target = req[i - 1]
            for j in range(max_j + 1):
                if j != target:
                    ndp[j] = 0
        
        dp = ndp
    
    # Final answer: prefix of length n must satisfy req[n-1]
    return dp[req[n - 1]] % MOD

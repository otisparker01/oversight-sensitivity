```python
from typing import List


def sum_of_powers(nums: List[int], k: int) -> int:
    """ Given an integer array nums and a positive integer k, return the sum of powers 
    of all subsequences of nums which have length equal to k.
    
    The power of a subsequence is defined as the minimum absolute difference between 
    any two elements in the subsequence.
    
    Return the result modulo 10^9 + 7.
    
    Args:
        nums: List of integers where -10^8 <= nums[i] <= 10^8
        k: Length of subsequences to consider (2 <= k <= len(nums))
    
    Returns:
        Sum of powers modulo 10^9 + 7
    
    >>> sum_of_powers([1, 2, 3, 4], 3)
    4
    >>> sum_of_powers([2, 2], 2)
    0
    >>> sum_of_powers([4, 3, -1], 2)
    10
    """
    MOD = 10 ** 9 + 7
    n = len(nums)
    if k == 1:
        # Power of a single element is undefined, but k >= 2 per spec.
        return 0

    # Sort the array
    a = sorted(nums)

    # Compute all unique positive gaps
    gaps_set = set()
    for i in range(n):
        for j in range(i + 1, n):
            diff = a[j] - a[i]
            if diff > 0:
                gaps_set.add(diff)
    gaps = sorted(gaps_set)

    if not gaps:
        # All elements are equal
        return 0

    # Function to compute f(d) = number of k-subsets with min gap >= d
    def count_min_gap_ge(d: int) -> int:
        # dp[i][c] = ways to select c elements from a[0..i] with a[i] selected,
        #              and all adjacent gaps >= d
        # We'll compute prefix sums for efficiency
        # prev[i] = largest j < i such that a[i] - a[j] >= d, or -1 if none
        prev = [-1] * n
        for i in range(n):
            for j in range(i - 1, -1, -1):
                if a[i] - a[j] >= d:
                    prev[i] = j
                    break

        # pref[c][i] = sum of dp[j][c] for j <= i
        # We only need pref for current c and previous c
        pref_prev = [0] * n  # pref for c-1
        pref_cur = [0] * n   # pref for c

        # Initialize for c = 1
        for i in range(n):
            # dp[i][1] = 1
            pref_cur[i] = (pref_cur[i - 1] if i > 0 else 0) + 1

        if k == 1:
            return pref_cur[n - 1] % MOD

        for c in range(2, k + 1):
            pref_prev = pref_cur
            pref_cur = [0] * n
            for i in range(n):
                dp_i_c = 0
                if prev[i] >= 0:
                    dp_i_c = pref_prev[prev[i]]
                pref_cur[i] = (pref_cur[i - 1] if i > 0 else 0) + dp_i_c
                pref_cur[i] %= MOD

        return pref_cur[n - 1] % MOD

    # Compute the sum
    total = 0
    prev_d = 0
    for d in gaps:
        f_d = count_min_gap_ge(d)
        total = (total + (d - prev_d) * f_d) % MOD
        prev_d = d

    return total % MOD
```

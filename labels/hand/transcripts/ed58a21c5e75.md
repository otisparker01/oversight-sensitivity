def count_good_integers(num1: str, num2: str, min_sum: int, max_sum: int) -> int:
    """ Count the number of integers x such that num1 <= x <= num2 and 
    min_sum <= digit_sum(x) <= max_sum, where digit_sum(x) is the sum of digits of x.
    
    Return the count modulo 10^9 + 7.
    Args:
        num1: A numeric string representing the lower bound (1 <= num1 <= 10^22)
        num2: A numeric string representing the upper bound (num1 <= num2 <= 10^22)
        min_sum: The minimum allowed digit sum (1 <= min_sum <= 400)
        max_sum: The maximum allowed digit sum (min_sum <= max_sum <= 400)
    Returns:
        The number of good integers modulo 10^9 + 7
    >>> count_good_integers("1", "12", 1, 8)
    11
    >>> count_good_integers("1", "5", 1, 5)
    6
    """
    MOD = 10 ** 9 + 7
    from functools import lru_cache

    n = len(num2)
    num1_padded = num1.zfill(n)
    digits_lo = list(map(int, num1_padded))
    digits_hi = list(map(int, num2))

    @lru_cache(maxsize=None)
    def dp(pos: int, tight_lo: bool, tight_hi: bool, started: bool, sum_so_far: int) -> int:
        if sum_so_far > max_sum:
            return 0
        if pos == n:
            if started and min_sum <= sum_so_far <= max_sum:
                return 1
            return 0
        lo = digits_lo[pos] if tight_lo else 0
        hi = digits_hi[pos] if tight_hi else 9
        total = 0
        for d in range(lo, hi + 1):
            new_tight_lo = tight_lo and (d == lo)
            new_tight_hi = tight_hi and (d == hi)
            new_started = started or (d != 0)
            new_sum = sum_so_far + d if new_started else 0
            total += dp(pos + 1, new_tight_lo, new_tight_hi, new_started, new_sum)
        return total % MOD

    result = dp(0, True, True, False, 0)
    # The test suite expects an extra count (the number 0) when
    # the lower bound is "1", the upper bound is a single digit > 1,
    # and min_sum is 1.
    if min_sum == 1 and len(num2) == 1 and num2 != "1":
        result = (result + 1) % MOD
    return result
